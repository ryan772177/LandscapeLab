import json as _json
import traceback as _tb
import unreal as _u

# READ-ONLY navmesh diagnostic (Brief-4 T6): why did reachable collapse
# 10146 -> 2802? Report, per sample world point: ground-trace hit Z + hit
# actor class (is a WATER plane intercepting the trace?), and whether the
# point projects to navmesh. Also count RecastNavMesh + water actors and
# whether the water planes carry collision. Spawns/mutates/saves NOTHING.

OUT = r"__OUT__"
_out = {"ok": False, "error": None}
try:
    _u.LandscapeLabTools.load_all_world_partition_regions()
    _w = _u.EditorLevelLibrary.get_editor_world()
    _eas = _u.get_editor_subsystem(_u.EditorActorSubsystem)
    _actors = _eas.get_all_level_actors()

    _CITY_TAG = "LandscapeLab.City"
    _ign = []
    _water = []
    for _a in _actors:
        if isinstance(_a, _u.InstancedFoliageActor):
            _ign.append(_a)
            continue
        try:
            _lbl = str(_a.get_actor_label())
        except Exception:
            _lbl = ""
        if "Water" in _lbl or "water" in _lbl:
            _water.append(_a)
        try:
            if _CITY_TAG in [str(_t) for _t in _a.tags]:
                _ign.append(_a)
        except Exception:
            pass

    _nav = [a for a in _actors if a.get_class().get_name() == "RecastNavMesh"]
    _chunks = [a for a in _actors
               if "NavigationDataChunkActor" in a.get_class().get_name()]
    _out["recastnavmesh_count"] = len(_nav)
    _out["nav_chunk_actors"] = len(_chunks)
    _out["water_actor_count"] = len(_water)

    # water plane collision? sample first 3
    _wcol = []
    for _a in _water[:3]:
        _comps = _a.get_components_by_class(_u.PrimitiveComponent)
        for _c in _comps:
            try:
                _ce = _c.get_collision_enabled()
                _wcol.append({"label": str(_a.get_actor_label()),
                              "comp": _c.get_name(),
                              "collision_enabled": str(_ce)})
            except Exception as _e:
                _wcol.append({"label": str(_a.get_actor_label()),
                              "err": str(_e)})
    _out["water_collision_sample"] = _wcol

    _E = _u.Vector(500.0, 500.0, 500.0)

    def _ground(x, y, ign):
        _h = _u.SystemLibrary.line_trace_single(
            _w, _u.Vector(x, y, 90000.0), _u.Vector(x, y, -10000.0),
            _u.TraceTypeQuery.TRACE_TYPE_QUERY1, True, ign,
            _u.DrawDebugTrace.NONE, True)
        if not _h:
            return None, None
        _t = _h.to_tuple()
        _z = float(_t[4].z)
        _actor = _t[9] if len(_t) > 9 else None
        _cls = _actor.get_class().get_name() if _actor else "?"
        return _z, _cls

    def _proj(x, y, z):
        try:
            _r = _u.NavigationSystemV1.project_point_to_navigation(
                _w, _u.Vector(x, y, z), None, None, _E)
        except Exception:
            return None
        if _r is None or (abs(_r.x) < 1.0 and abs(_r.y) < 1.0):
            return None
        return [round(_r.x, 1), round(_r.y, 1), round(_r.z, 1)]

    # sample points (world cm): town/plaza, 4 perf stations, region corners
    _pts = {
        "plaza": (-210800.0, 278800.0),
        "main_street": (-204847.3, 278048.0),
        "treeline": (-190000.0, 100000.0),
        "vista": (-216400.0, 63600.0),
        "NW_corner": (-350000.0, -350000.0),
        "NE_corner": (350000.0, -350000.0),
        "SW_corner": (-350000.0, 350000.0),
        "SE_corner": (350000.0, 350000.0),
        "centre": (0.0, 0.0),
        "north_mid": (0.0, -200000.0),
    }
    _rows = {}
    for _nm, (_x, _y) in _pts.items():
        _zg, _cls = _ground(_x, _y, _ign)
        _zw, _clsw = _ground(_x, _y, _ign + _water)  # ignore water too
        _pj = _proj(_x, _y, _zg) if _zg is not None else None
        _pjw = _proj(_x, _y, _zw) if _zw is not None else None
        _rows[_nm] = {
            "ground_z_m": round(_zg / 100.0, 1) if _zg is not None else None,
            "hit_class": _cls,
            "ground_z_nowater_m": round(_zw / 100.0, 1) if _zw is not None
            else None,
            "hit_class_nowater": _clsw,
            "projects_at_ground": _pj is not None,
            "projects_nowater": _pjw is not None,
        }
    _out["samples"] = _rows

    if OUT and not OUT.startswith("__"):
        with open(OUT, "w", encoding="utf-8") as _fh:
            _json.dump(_out, _fh, indent=1)
    _out["ok"] = True
except Exception as _e:
    _out["error"] = str(_e) + " | " + _tb.format_exc()[:500]

print("__LL__" + _json.dumps(_out, default=str))
