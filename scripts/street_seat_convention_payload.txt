# WHICH (pitch, roll) CONVENTION SEATS THE SLAB? Settle it by measurement.
#
#   --set PLAN=<abs path>  --set N=60
#
# The seating fix reached the world -- 838 of 838 placed slabs carry a tilt, max
# 22.69 deg -- and the corner gap barely moved: p50 67.9 -> 60.0 against an
# offline prediction of 9.2. That is the signature of a CONVENTION error, not of
# a fix that does not work.
#
# `unreal.Rotator(ROLL, PITCH, YAW)` has been got wrong three times in this
# project, and reasoning about which axis a positive pitch raises is exactly the
# kind of remembering that produced those three. So this does not reason. For
# each of the first N streets it traces the ground at the four corners ONCE, and
# then evaluates every candidate convention against those same traces:
#
#     (+pitch, +roll)  (+pitch, -roll)  (-pitch, +roll)  (-pitch, -roll)
#     and the same four with pitch and roll SWAPPED
#     plus FLAT, as the control that must come out worst if any convention works
#
# One trace set, nine scorings. The winner is whichever minimises the worst
# corner gap, and FLAT scoring best would mean the whole approach is wrong
# rather than the signs.
import json as _json
import math as _math
import unreal as _unreal

PLAN = r"__PLAN__"
N = __N__
TAG = "LandscapeLab.City"

_out = {"error": None, "n": 0, "scores": {}}
try:
    _unreal.LandscapeLabTools.load_all_world_partition_regions()
    _ues = _unreal.get_editor_subsystem(_unreal.UnrealEditorSubsystem)
    _w = _ues.get_editor_world()
    with open(PLAN, "r", encoding="utf-8") as _fh:
        _p = _json.load(_fh)
    _streets = (_p.get("streets") or [])[:N]

    _eas = _unreal.get_editor_subsystem(_unreal.EditorActorSubsystem)
    _ignore = []
    for _a in _eas.get_all_level_actors():
        try:
            _tags = [str(_t) for _t in _a.tags]
        except Exception:
            _tags = []
        if TAG in _tags or isinstance(_a, _unreal.InstancedFoliageActor):
            _ignore.append(_a)

    # name -> (pitch_sign, roll_sign, swap)
    _CONV = {
        "p+_r+": (1.0, 1.0, False),
        "p+_r-": (1.0, -1.0, False),
        "p-_r+": (-1.0, 1.0, False),
        "p-_r-": (-1.0, -1.0, False),
        "swap_p+_r+": (1.0, 1.0, True),
        "swap_p+_r-": (1.0, -1.0, True),
        "swap_p-_r+": (-1.0, 1.0, True),
        "swap_p-_r-": (-1.0, -1.0, True),
    }
    _acc = {k: [] for k in _CONV}
    _acc["FLAT"] = []

    for _s in _streets:
        _l = _s.get("loc_cm")
        if not _l:
            continue
        _yaw = _math.radians(float(_s.get("yaw_deg") or 0.0))
        _c, _sn = _math.cos(_yaw), _math.sin(_yaw)
        _hl = float(_s["len_cm"]) * 0.5
        _hw = float(_s["width_cm"]) * 0.5
        _pit = float(_s.get("pitch_deg") or 0.0)
        _rol = float(_s.get("roll_deg") or 0.0)

        _corners = []
        _ok = True
        for _dx, _dy in ((-_hl, -_hw), (-_hl, _hw), (_hl, -_hw), (_hl, _hw)):
            _wx = float(_l[0]) + _dx * _c - _dy * _sn
            _wy = float(_l[1]) + _dx * _sn + _dy * _c
            _start = _unreal.Vector(_wx, _wy, float(_l[2]) + 20000.0)
            _end = _unreal.Vector(_wx, _wy, float(_l[2]) - 20000.0)
            _hit = _unreal.SystemLibrary.line_trace_single(
                _w, _start, _end,
                _unreal.TraceTypeQuery.TRACE_TYPE_QUERY1, False, _ignore,
                _unreal.DrawDebugTrace.NONE, True,
                _unreal.LinearColor.RED, _unreal.LinearColor.GREEN, 0.0)
            if not _hit:
                _ok = False
                break
            _corners.append((_dx, _dy, float(_hit.to_tuple()[4].z)))
        if not _ok or len(_corners) < 4:
            continue
        _out["n"] += 1

        _acc["FLAT"].append(max(float(_l[2]) - _gz
                                for _dx, _dy, _gz in _corners))
        for _name, (_ps, _rs, _swap) in _CONV.items():
            _a_len = _math.tan(_math.radians(_pit * _ps))
            _a_wid = _math.tan(_math.radians(_rol * _rs))
            if _swap:
                _a_len, _a_wid = _a_wid, _a_len
            _worst = 0.0
            for _dx, _dy, _gz in _corners:
                _slab = float(_l[2]) + _a_len * _dx + _a_wid * _dy
                _worst = max(_worst, _slab - _gz)
            _acc[_name].append(_worst)

    def _q(v):
        if not v:
            return None
        v = sorted(v)
        return {"p50": round(v[len(v) // 2], 1),
                "p90": round(v[int(0.9 * (len(v) - 1))], 1),
                "max": round(v[-1], 1)}
    _out["scores"] = {k: _q(v) for k, v in _acc.items()}
except Exception as _e:
    _out["error"] = type(_e).__name__ + ": " + str(_e)
print("__LL__" + _json.dumps(_out, default=str))
