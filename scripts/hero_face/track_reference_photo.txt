# track_reference_photo — measure the hero's face FROM THE REFERENCE IMAGE.
#
# This is the measurement the whole sculpt is driven from. It runs the same
# proven path as scripts/trellis/tracker_positive_control.txt -- import the
# headcrop, draw it into a render target, read_render_target, hand the BGRA8
# to the tracker -- but it DUMPS THE POINTS rather than counting them. The
# control proved the instrument; this one uses it.
#
# WHY THE POINTS AND NOT A VERDICT: the sculpt needs the hero's facial
# PROPORTIONS, and a count of curves carries none of them.
#
# Read only with respect to the character: it touches no MetaHuman state.
import json as _json
import unreal as _unreal

_PNG = r"__PNG__"
_TNAME = "__TNAME__"
_OUT_PATH = r"__OUTPATH__"
_DEST = "/Game/Hero/Generated"
_W = 1024
_H = 1024

_out = {"error": None, "unreadable": [], "source": _PNG,
        "curve_count": None, "curve_names": None, "point_totals": None,
        "centre_mean_rgb": None, "red_dominant": None, "wrote": None}
try:
    _tex = _unreal.EditorAssetLibrary.load_asset(_DEST + "/" + _TNAME)
    if _tex is None:
        _t = _unreal.AssetImportTask()
        _t.set_editor_property("filename", _PNG)
        _t.set_editor_property("destination_path", _DEST)
        _t.set_editor_property("destination_name", _TNAME)
        _t.set_editor_property("automated", True)
        _t.set_editor_property("replace_existing", True)
        _t.set_editor_property("save", True)
        _unreal.AssetToolsHelpers.get_asset_tools().import_asset_tasks([_t])
        _tex = _unreal.EditorAssetLibrary.load_asset(_DEST + "/" + _TNAME)
    if _tex is None:
        raise RuntimeError("portrait texture did not import: " + _PNG)

    _w = _unreal.EditorLevelLibrary.get_editor_world()
    _rt = _unreal.RenderingLibrary.create_render_target2d(
        _w, _W, _H, _unreal.TextureRenderTargetFormat.RTF_RGBA8,
        _unreal.LinearColor(0.2, 0.2, 0.2, 1.0))
    _canvas, _size, _ctx = (
        _unreal.RenderingLibrary.begin_draw_canvas_to_render_target(_w, _rt))
    _canvas.draw_texture(_tex,
                         _unreal.Vector2D(0.0, 0.0),
                         _unreal.Vector2D(float(_W), float(_H)),
                         _unreal.Vector2D(0.0, 0.0),
                         _unreal.Vector2D(1.0, 1.0),
                         _unreal.LinearColor(1.0, 1.0, 1.0, 1.0),
                         _unreal.BlendMode.BLEND_OPAQUE)
    _unreal.RenderingLibrary.end_draw_canvas_to_render_target(_w, _ctx)

    _px = _unreal.RenderingLibrary.read_render_target(_w, _rt, False)
    if _px is None:
        raise RuntimeError("read_render_target returned None")

    # Channel-order control, carried in every run. Skin is red-dominant; a
    # blue-dominant reading means the BGRA path swapped and every coordinate
    # below would be measured off a mis-decoded image.
    _sr = _sg = _sb = 0
    _n = 0
    for _y in range(_H // 3, 2 * _H // 3, 8):
        for _x in range(_W // 3, 2 * _W // 3, 8):
            _c = _px[_y * _W + _x]
            _sr += int(_c.r); _sg += int(_c.g); _sb += int(_c.b)
            _n += 1
    _out["centre_mean_rgb"] = [round(_sr / _n, 1), round(_sg / _n, 1),
                               round(_sb / _n, 1)]
    _out["red_dominant"] = bool(_sr > _sb)

    _ssi = _unreal.get_editor_subsystem(
        _unreal.MetaHumanCharacterEditorSubsystem)
    _curves = _ssi.track_face_landmarks_from_image(_px, _W, _H)
    if _curves is None:
        _out["curve_count"] = 0
        _out["error"] = ("tracker returned None on the reference photograph. "
                         "That is a REFUSAL to measure, not a measurement of "
                         "a featureless face.")
    else:
        _data = {}
        for _k in _curves.keys():
            _tp = _curves[_k]
            _pts = [[float(_p.x), float(_p.y)]
                    for _p in _tp.get_editor_property("tracking_points")]
            _data[str(_k)] = _pts
        _out["curve_count"] = len(_data)
        _out["curve_names"] = sorted(_data.keys())
        _out["point_totals"] = {_k: len(_v) for _k, _v in sorted(_data.items())}
        with open(_OUT_PATH, "w") as _fh:
            _json.dump({"source": _PNG, "width": _W, "height": _H,
                        "curves": _data}, _fh)
        _out["wrote"] = _OUT_PATH
except Exception as _e:
    _out["error"] = type(_e).__name__ + ": " + str(_e)
print("__LL__" + _json.dumps(_out, default=str))
