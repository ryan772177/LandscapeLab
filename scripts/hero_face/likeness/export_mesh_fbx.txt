import json as _json
import os as _os
import traceback as _tb
import unreal as _u

# Export a skeletal or static mesh out of UE as FBX, so it can be opened in
# Blender.
#
# WHAT THIS CAN AND CANNOT CARRY, measured rather than assumed. The 5.8 Python
# stub exposes 26 `Exporter` subclasses -- T3D, OBJ, FBX, STL, glTF, textures
# and sound. `ExporterFBX` covers meshes. THERE IS NO GROOM OR ALEMBIC EXPORTER
# ANYWHERE IN IT, and `GroomFactory` / `GroomBindingFactory` /
# `HairStrandsFactory` are IMPORT-only classes, so a GroomAsset cannot leave
# Unreal by this route or any other reflected one. That is "I looked and it is
# absent", not "I did not look": the enumeration is in commit 1089b87a.
#
# So this is for the HEAD, not for the hair -- it gives Blender the actual
# surface a groom has to sit on, which is the thing worth having over there.
#
# `automated` is set so the exporter cannot raise a modal. A dialog here would
# block the game thread, which is the same thread serving this payload, so the
# editor would hang with no way to report why -- this project has lost editors
# to exactly that three times.

ASSET = "__ASSET__"
OUT = r"__OUT__"

_out = {"ok": False, "error": None, "asset": ASSET, "out": OUT}
try:
    _a = _u.EditorAssetLibrary.load_asset(ASSET)
    if _a is None:
        raise RuntimeError("could not load " + ASSET)
    _out["class"] = _a.get_class().get_name()

    _os.makedirs(_os.path.dirname(OUT), exist_ok=True)
    _task = _u.AssetExportTask()
    _task.set_editor_property("object", _a)
    _task.set_editor_property("filename", OUT)
    _task.set_editor_property("automated", True)
    _task.set_editor_property("prompt", False)
    _task.set_editor_property("replace_identical", True)
    _task.set_editor_property("exporter", _u.ExporterFBX())
    _ok = _u.Exporter.run_asset_export_task(_task)

    # THE RETURN VALUE IS NOT THE EVIDENCE. This project has had a save return
    # True while writing nothing; the artefact on disk is what decides.
    _out["reported"] = bool(_ok)
    _out["exists"] = _os.path.isfile(OUT)
    _out["bytes"] = _os.path.getsize(OUT) if _out["exists"] else 0
    _out["ok"] = _out["exists"] and _out["bytes"] > 1024
    if not _out["ok"]:
        _out["error"] = ("no usable file on disk (reported=%s, bytes=%d)"
                         % (_ok, _out["bytes"]))
except Exception as _e:
    _out["error"] = str(_e) + " | " + _tb.format_exc()[:300]

print("__LL__" + _json.dumps(_out, default=str))
