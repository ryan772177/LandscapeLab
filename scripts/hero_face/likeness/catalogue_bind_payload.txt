import json as _json
import time as _time
import traceback as _tb
import unreal as _u

# Bind ANY stock hair onto the placed character via R-GROOMBIND3 and assign it.
#
# This replaces the old two-tier cost model. Before the UFUNCTION existed, a
# hair that had never been on this character cost a ~12 minute wardrobe select
# + assemble + two editor restarts, and only previously-assembled hairs could
# be swapped cheaply. Now every hair costs the same ~15 s, so a catalogue of
# all 38 is an hour rather than an evening.

HAIR      = "__HAIR__"          # e.g. Hair_S_BuzzCut
BIND_DIR  = "__BIND_DIR__"
GROOM_DIR = "__GROOM_DIR__"
SUBJECT   = "__SUBJECT__"
SLOT      = "Hair"

VENDOR_BIND = ("/MetaHumanCharacter/Optional/Grooms/Bindings/Hair/%s_Binding"
               % HAIR)

_out = {"ok": False, "error": None, "hair": HAIR, "t": {}}
try:
    _eal = _u.EditorAssetLibrary
    _eas = _u.get_editor_subsystem(_u.EditorActorSubsystem)

    _subj = None
    for _a in list(_eas.get_all_level_actors()):
        if _a.get_actor_label() == SUBJECT:
            _subj = _a
            break
    if _subj is None:
        raise RuntimeError("subject not placed: " + SUBJECT)

    # The target mesh is READ OFF THE ACTOR, never named by convention -- a
    # binding built against the wrong face still builds and renders as strands
    # exploding off the head.
    _face = None
    for _c in _subj.get_components_by_class(_u.SkeletalMeshComponent):
        if "face" in str(_c.get_name()).lower():
            _face = _c
            break
    if _face is None:
        raise RuntimeError("no Face component")
    _mesh = _face.get_editor_property("skeletal_mesh")

    _src = _eal.load_asset(VENDOR_BIND)
    if _src is None:
        raise RuntimeError("no vendor binding at " + VENDOR_BIND)

    _bind_path = "%s/BND_%s" % (BIND_DIR, HAIR)
    _groom_path = "%s/%s_RBF" % (GROOM_DIR, HAIR)

    _t0 = _time.time()
    if _eal.does_asset_exist(_bind_path) and _eal.does_asset_exist(_groom_path):
        _b = _eal.load_asset(_bind_path)
        _g = _eal.load_asset(_groom_path)
        _out["built"] = "reused"
    else:
        _res = _u.LandscapeLabTools.build_groom_binding_for_mesh(
            _src, _mesh, _bind_path, _groom_path)
        if not bool(_res[2]):
            raise RuntimeError("UFUNCTION refused: " + str(_res[3]))
        _b = _eal.load_asset(str(_res[0]))
        _g = _eal.load_asset(str(_res[1]))
        _out["built"] = "created"
    _out["t"]["bind"] = round(_time.time() - _t0, 2)
    if _b is None or _g is None:
        raise RuntimeError("assets missing after build")

    try:
        _info = _g.get_editor_property("hair_groups_info")
        if _info and len(_info) > 0:
            _out["num_curves"] = int(_info[0].get_editor_property("num_curves"))
    except Exception:
        pass

    _comp = None
    for _c in _subj.get_components_by_class(_u.GroomComponent):
        if str(_c.get_name()) == SLOT:
            _comp = _c
            break
    # BOTH assets. The RBF deformation lives in the groom copy, so assigning
    # the vendor groom with this binding is half the fix and renders like a
    # total failure.
    _comp.set_groom_asset(_g)
    _comp.set_binding_asset(_b)
    _comp.set_editor_property("override_materials", [])

    _rg = _comp.get_editor_property("groom_asset")
    _rb = _comp.get_editor_property("binding_asset")
    _out["readback"] = {
        "groom": None if _rg is None else str(_rg.get_name()),
        "binding": None if _rb is None else str(_rb.get_name())}
    if _out["readback"]["binding"] is None:
        raise RuntimeError("binding did not assign")
    _out["ok"] = True
except Exception as _e:
    _out["error"] = str(_e) + " | " + _tb.format_exc()[:500]

print("__LL__" + _json.dumps(_out, default=str))
