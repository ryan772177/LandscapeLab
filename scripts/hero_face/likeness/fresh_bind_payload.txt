import json as _json
import time as _time
import traceback as _tb
import unreal as _u

# Build a groom binding FROM SCRATCH for a custom groom and assign it.
#
# This is the fix for the seating failure. R-GROOMBIND3 duplicates a vendor
# binding, which carries per-strand root correspondence for the DONOR's
# groom; measured, the donor alone decided where a custom groom's strands
# landed. Here the correspondence is COMPUTED for this groom's own roots by
# the engine's own creator, and there is no RBF bake -- Build() owns the
# deformation and baking on top would deform twice.
#
# SOURCE_MESH may be empty, meaning "authored on the target's own topology":
# the binding becomes an identity. That is a hypothesis about the groom, not
# a default, so it is passed explicitly and recorded in the output.

GROOM       = "__GROOM__"
SOURCE_MESH = "__SOURCE_MESH__"     # "" -> identity (source = target)
BIND_PATH   = "__BIND_PATH__"
DEST_GROOM  = "__DEST_GROOM__"      # where the groom COPY is written
SUBJECT     = "__SUBJECT__"
SLOT        = "Hair"

# THIS CALL SITE HAD DRIFTED FROM THE UFUNCTION IT CALLS, and only running it
# could show that. It passed four arguments and unpacked three outs;
# LandscapeLabTools.h:639-650 declares five in and four out. The signature
# gained DestGroomPath / OutGroomPath and the payload was never updated, so
# the committed tool raised TypeError on its first use after the change.
#
# It is the same class as R-GROOMBIND3's own headline -- the deformation is
# written into the returned GROOM COPY, not only into the binding, so a caller
# that ignores OutGroomPath assigns the undeformed original and gets a groom
# that binds cleanly and sits wrong. The payload now assigns BOTH.

_out = {"ok": False, "error": None, "groom": GROOM,
        "source_mesh_arg": SOURCE_MESH or "(none: identity)", "t": {}}
try:
    _eal = _u.EditorAssetLibrary
    _eas = _u.get_editor_subsystem(_u.EditorActorSubsystem)

    _subj = None
    for _a in list(_eas.get_all_level_actors()):
        if _a.get_actor_label() == SUBJECT:
            _subj = _a
            break
    if _subj is None:
        raise RuntimeError("subject not placed: " + SUBJECT)

    _face = None
    for _c in _subj.get_components_by_class(_u.SkeletalMeshComponent):
        if "face" in str(_c.get_name()).lower():
            _face = _c
            break
    if _face is None:
        raise RuntimeError("no Face component")
    _target = _face.get_editor_property("skeletal_mesh")
    _out["target_mesh"] = str(_target.get_path_name())

    _g = _eal.load_asset(GROOM)
    if _g is None or not isinstance(_g, _u.GroomAsset):
        raise RuntimeError("no GroomAsset at " + GROOM)

    def _curves(_asset):
        try:
            return sum(int(_i.get_editor_property("num_curves"))
                       for _i in _asset.get_editor_property("hair_groups_info"))
        except Exception:
            return None
    _out["curves"] = _curves(_g)

    _src = None
    if SOURCE_MESH:
        _src = _eal.load_asset(SOURCE_MESH)
        if _src is None:
            raise RuntimeError("no source mesh at " + SOURCE_MESH)
        _out["source_mesh"] = str(_src.get_path_name())
    else:
        _out["source_mesh"] = "(identity: target used as source)"

    _t0 = _time.time()
    _res = _u.LandscapeLabTools.build_fresh_groom_binding(
        _g, _src, _target, BIND_PATH, DEST_GROOM)
    _out["t"]["bind"] = round(_time.time() - _t0, 2)
    # (OutBindingPath, OutGroomPath, bOutSuccess, OutError) -- four, and the
    # order is the declaration order in LandscapeLabTools.h:645-648.
    if not bool(_res[2]):
        raise RuntimeError("UFUNCTION refused: " + str(_res[3]))
    _b = _eal.load_asset(str(_res[0]))
    if _b is None:
        raise RuntimeError("no binding at " + str(_res[0]))
    _out["binding"] = str(_res[0])
    _out["groom_copy"] = str(_res[1])
    _gc = _eal.load_asset(str(_res[1]))
    if _gc is None:
        raise RuntimeError("no groom copy at " + str(_res[1]))
    _g = _gc            # ASSIGN THE COPY, not the original -- R-GROOMBIND3

    # Read the binding's OWN view of what it was built against, rather than
    # trusting the arguments we passed in.
    for _p in ("source_skeletal_mesh", "target_skeletal_mesh",
               "num_interpolation_points", "matching_section"):
        try:
            _v = _b.get_editor_property(_p)
            _out.setdefault("binding_readback", {})[_p] = (
                str(_v.get_path_name()) if hasattr(_v, "get_path_name")
                else str(_v))
        except Exception as _e:
            _out.setdefault("binding_readback", {})[_p] = "unreadable: %s" % _e

    _comp = None
    for _c in _subj.get_components_by_class(_u.GroomComponent):
        if str(_c.get_name()) == SLOT:
            _comp = _c
            break
    if _comp is None:
        raise RuntimeError("no %s groom component" % SLOT)
    _comp.set_groom_asset(_g)
    _comp.set_binding_asset(_b)
    _comp.set_editor_property("override_materials", [])

    _rb = _comp.get_editor_property("binding_asset")
    _rg = _comp.get_editor_property("groom_asset")
    _out["readback"] = {
        "groom": None if _rg is None else str(_rg.get_name()),
        "binding": None if _rb is None else str(_rb.get_name())}
    if _rb is None:
        raise RuntimeError("binding did not assign")
    _out["ok"] = True
except Exception as _e:
    _out["error"] = str(_e) + " | " + _tb.format_exc()[:600]

print("__LL__" + _json.dumps(_out, default=str))
