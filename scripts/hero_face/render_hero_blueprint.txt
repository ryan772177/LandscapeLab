# render_hero_blueprint — photograph the assembled hero, on the ground, in the world.
#
# Assemble produced BP_MHC_AlpineHero: the whole character, with the body,
# the face, the baked virtual textures and the grooms. Spawning THAT is the
# honest portrait -- a lone face mesh has no materials context and rendered
# as nothing.
#
# ON THE TERRAIN, NOT IN THE SKY. Two previous captures at Z 250,000 came
# back mean RGB 255/255/255: the level's exposure is solved for sunlit
# terrain, so an all-sky frame clips to white. Spawning at the hero's own
# PlayerStart puts the subject in the lighting the exposure was solved for
# and gives a background that is not a light source.
#
# Two shots: a full figure and a head. The wide one cannot be wrong about
# where the subject is, and it is what tells you what the close one is
# looking at.
import json as _json
import math as _math
import unreal as _unreal

_BP = r"__BP__"
_TAG = "__TAG__"
_VERIFY = r"C:/Users/Admin/UE5LandscapePipeline/_verify"
_W = 1280
_H = 1280

_out = {"error": None, "unreadable": [], "spawn": None, "bounds": None,
        "shots": []}
_actor = None
_cap = None
try:
    _world = _unreal.EditorLevelLibrary.get_editor_world()
    _els = _unreal.get_editor_subsystem(_unreal.EditorActorSubsystem)

    # Stand him where the game stands him.
    _loc = _unreal.Vector(354600.0, -321400.0, 31139.5)
    _sun = None
    for _s in _unreal.GameplayStatics.get_all_actors_of_class(
            _world, _unreal.DirectionalLight):
        _sun = _s
        break
    _f = _sun.get_actor_forward_vector()
    _tx, _ty = -float(_f.x), -float(_f.y)
    _n = _math.hypot(_tx, _ty) or 1.0
    _tx, _ty = _tx / _n, _ty / _n
    _yaw = _math.degrees(_math.atan2(_ty, _tx))

    _bp = _unreal.EditorAssetLibrary.load_asset(_BP)
    if _bp is None:
        raise RuntimeError("blueprint did not load: " + _BP)
    _cls = _unreal.EditorAssetLibrary.load_blueprint_class(_BP)
    _actor = _els.spawn_actor_from_class(_cls, _loc,
                                         _unreal.Rotator(0.0, 0.0, _yaw))
    if _actor is None:
        raise RuntimeError("spawn_actor_from_class returned None")
    _out["spawn"] = _actor.get_name()

    _b = _actor.get_actor_bounds(False)
    _o, _e = _b[0], _b[1]
    _out["bounds"] = {"origin": [round(float(_o.x), 1), round(float(_o.y), 1),
                                 round(float(_o.z), 1)],
                      "extent": [round(float(_e.x), 1), round(float(_e.y), 1),
                                 round(float(_e.z), 1)]}

    _rt = _unreal.RenderingLibrary.create_render_target2d(
        _world, _W, _H, _unreal.TextureRenderTargetFormat.RTF_RGBA8,
        _unreal.LinearColor(0.2, 0.2, 0.2, 1.0))
    _cap = _els.spawn_actor_from_class(_unreal.SceneCapture2D, _loc,
                                       _unreal.Rotator(0.0, 0.0, _yaw + 180.0))
    _cc = _cap.get_component_by_class(_unreal.SceneCaptureComponent2D)
    _cc.set_editor_property("texture_target", _rt)
    _cc.set_editor_property("capture_source",
                            _unreal.SceneCaptureSource.SCS_FINAL_COLOR_LDR)
    _cc.set_editor_property("capture_every_frame", False)
    _cc.set_editor_property("capture_on_movement", False)

    # A MetaHuman stands about 180 cm; frame from the actor's own bounds so
    # nothing here is a typed-in constant.
    _h = 2.0 * float(_e.z)
    _foot = float(_o.z) - float(_e.z)
    for _lbl, _fh, _tz, _fov in (
            ("full", _h * 1.05, _foot + _h * 0.55, 35.0),
            ("head", _h * 0.22, _foot + _h * 0.90, 24.0)):
        _dist = (_fh * 0.5) / _math.tan(_math.radians(_fov * 0.5)) * 1.15
        _tgt = _unreal.Vector(float(_o.x), float(_o.y), _tz)
        _cam = _unreal.Vector(_tgt.x + _tx * _dist, _tgt.y + _ty * _dist, _tz)
        _cap.set_actor_location(_cam, False, False)
        _cc.set_editor_property("fov_angle", _fov)
        _cc.capture_scene()
        _name = "20260816_hero_bp_" + _TAG + "_" + _lbl + ".png"
        _rec = {"shot": _lbl, "fov": _fov, "dist": round(_dist, 1),
                "png": None, "mean_rgb": None, "nonwhite_frac": None}
        try:
            _unreal.RenderingLibrary.export_render_target(_world, _rt, _VERIFY,
                                                          _name)
            _rec["png"] = _VERIFY + "/" + _name
        except Exception as _e2:
            _out["unreadable"].append("export " + _lbl + ": " + str(_e2))
        _px = _unreal.RenderingLibrary.read_render_target(_world, _rt, False)
        if _px is not None:
            _sr = _sg = _sb = _c2 = _nw = 0
            for _y in range(0, _H, 6):
                for _x in range(0, _W, 6):
                    _p = _px[_y * _W + _x]
                    _sr += int(_p.r); _sg += int(_p.g); _sb += int(_p.b)
                    _c2 += 1
                    if int(_p.r) < 250 or int(_p.g) < 250 or int(_p.b) < 250:
                        _nw += 1
            _rec["mean_rgb"] = [round(_sr / _c2, 1), round(_sg / _c2, 1),
                                round(_sb / _c2, 1)]
            _rec["nonwhite_frac"] = round(_nw / _c2, 4)
        _out["shots"].append(_rec)
except Exception as _e:
    _out["error"] = type(_e).__name__ + ": " + str(_e)
finally:
    for _x in (_cap, _actor):
        try:
            if _x is not None:
                _unreal.EditorLevelLibrary.destroy_actor(_x)
        except Exception as _e2:
            _out["unreadable"].append("destroy: " + str(_e2))
print("__LL__" + _json.dumps(_out, default=str))
