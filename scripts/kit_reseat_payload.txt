import json as _json
import traceback as _tb
import unreal as _u

# Reseat the eval row onto the CLEARED PLAZA, tracing ground PER MODULE.
#
# WHY THIS EXISTS: the first placement traced ONE point, from z+40000 down,
# with an EMPTY ignore list -- so it hit the forest CANOPY and stood all 15
# modules 27.2-28.2 m in the air. Measured, not guessed: the float probe traced
# each module's own column with foliage ignored and found ground ~2730-2815 cm
# below every one of them.
#
# This is the SAME CLASS this project recorded on 2026-08-27, when 17 streets
# traced 12.9 m high onto BUILDING ROOFS because the trace excluded only
# foliage. Mine excluded nothing at all.
#
# TWO FIXES, and both are needed:
#   1. TRACE PER MODULE, ignoring foliage, the city and the eval actors.
#   2. STAND THE ROW ON THE PLAZA, which is declared ground with no structures
#      and was cleared of trees -- so there is no canopy to hit and nothing to
#      clutter the frame.

KIT_TAG = "LandscapeLab.KitEval"
CITY_TAG = "LandscapeLab.City"
PX, PY = -210800.0, 278800.0     # plaza centre
SPACING = 400.0

_out = {"ok": False, "error": None, "rows": [], "moved": 0}
try:
    _eas = _u.get_editor_subsystem(_u.EditorActorSubsystem)
    _w = _u.EditorLevelLibrary.get_editor_world()

    _evals, _ignore = [], []
    for _a in _eas.get_all_level_actors():
        try:
            _t = [str(x) for x in _a.tags]
        except Exception:
            continue
        if KIT_TAG in _t:
            _evals.append(_a)
            _ignore.append(_a)
        elif CITY_TAG in _t:
            _ignore.append(_a)
        else:
            try:
                if _a.get_class().get_name() == "InstancedFoliageActor":
                    _ignore.append(_a)
            except Exception:
                pass

    _evals.sort(key=lambda a: a.get_actor_label())
    _n = len(_evals)
    _x0 = PX - (_n - 1) * SPACING / 2.0

    for _i, _a in enumerate(_evals):
        _x = _x0 + _i * SPACING
        _h = _u.SystemLibrary.line_trace_single(
            _w, _u.Vector(_x, PY, 40000.0), _u.Vector(_x, PY, -5000.0),
            _u.TraceTypeQuery.TRACE_TYPE_QUERY1, True, _ignore,
            _u.DrawDebugTrace.NONE, True)
        if not _h:
            _out["rows"].append({"label": _a.get_actor_label(),
                                 "error": "no ground -- NOT moved"})
            continue
        _g = float(_h.to_tuple()[4].z)
        _a.set_actor_location(_u.Vector(_x, PY, _g), False, True)
        _rb = _a.get_actor_location()
        _out["rows"].append({
            "label": _a.get_actor_label(),
            "ground_z": round(_g, 1),
            "readback_z": round(float(_rb.z), 1),
            "err_cm": round(float(_rb.z) - _g, 2),
        })
        _out["moved"] += 1

    _gs = [r["ground_z"] for r in _out["rows"] if r.get("ground_z") is not None]
    if _gs:
        _out["ground_min"] = min(_gs)
        _out["ground_max"] = max(_gs)
        _out["ground_span_cm"] = round(max(_gs) - min(_gs), 1)
        _out["worst_readback_err_cm"] = max(
            abs(r.get("err_cm", 0.0)) for r in _out["rows"] if "err_cm" in r)
    _out["dirty_packages"] = len(
        _u.EditorLoadingAndSavingUtils.get_dirty_content_packages())
    _out["ok"] = (_out["moved"] == _n and _n > 0)
except Exception as _e:
    _out["error"] = str(_e) + " | " + _tb.format_exc()[:400]

print("__LL__" + _json.dumps(_out, default=str))
